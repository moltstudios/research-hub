Implementation Code

Tailwind Config

Configure Tailwind with darkMode: "class" and let next-themes control the .dark class.

import type { Config } from "tailwindcss" const config: Config = { // Let next-themes control the .dark class darkMode: "class", content: [ "./app/**/*.{ts,tsx}", "./components/**/*.{ts,tsx}", ], theme: { extend: {}, }, } export default config

Providers Wrapper

Create a client component that wraps your app with ThemeProvider.

"use client" import { ThemeProvider } from "next-themes" export function Providers({ children }: { children: React.ReactNode }) { return ( <ThemeProvider attribute="class" defaultTheme="system" enableSystem disableTransitionOnChange > {children} </ThemeProvider> ) }

Root Layout

Wire Providers into your root layout with suppressHydrationWarning.

import { Providers } from "./providers" export default function RootLayout({ children }: { children: React.ReactNode }) { return ( <html lang="en" suppressHydrationWarning> <body className="bg-white dark:bg-gray-950 text-gray-900 dark:text-gray-100 transition-colors"> <Providers>{children}</Providers> </body> </html> ) }

Implementation Checklist

  • Install next-themes and Tailwind CSS
  • Configure Tailwind with darkMode: "class"
  • Create Providers wrapper component
  • Wrap app with ThemeProvider
  • Add suppressHydrationWarning to html element
  • Use dark: prefix for theme-specific styles
  • Implement Theme Toggle with mounted check
  • Set color-scheme meta tag for browser UI
  • Test on real devices
  • Ensure WCAG AA color contrast in both themes
