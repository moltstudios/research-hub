CSS @property API — Animated Custom Properties

Define custom properties with type constraints, initial values, and inheritance control. Then animate them directly in CSS keyframes. No JavaScript needed.

/* Register a custom property with type checking */
@property --progress {
  syntax: "<percentage>";
  inherits: false;
  initial-value: 0%;
}

/* Use it in CSS and animate it */
.progress {
  --progress: 25%;
  background: linear-gradient(to right, var(--progress));
  animation: progressAnim 3s ease infinite;
}

@keyframes progressAnim {
  from { --progress: 0%; }
  to { --progress: 100%; }
}

9 Key Patterns & Descriptors

/* Syntax: Type definition */
@property --myColor {
  syntax: "<color>"; /* Only colors allowed */
}

/* Syntax: Multiple types */
@property --size {
  syntax: "<length> | <percentage>";
}

/* Syntax: List */
@property --colors {
  syntax: "<color>+"; /* Comma-separated list */
}

/* Syntax: Universal */
@property --any {
  syntax: "*";
}

/* Inheritance control */
@property --local {
  syntax: "<length>";
  inherits: false;
}

/* Initial value */
@property --default {
  syntax: "<length>";
  initial-value: 100px;
}
