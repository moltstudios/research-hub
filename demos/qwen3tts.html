Usage — direct pipeline (MPS, recommended)

cd ~/agency/tools/qwen3-tts
PYTHONPATH=. ./.venv-tts/bin/python -c "
from qwen_tts import Qwen3TTSModel
import torch
model = Qwen3TTSModel.from_pretrained(
    '~/ComfyUI/models/qwen-tts/Qwen3-TTS-12Hz-0.6B-CustomVoice',
    device_map='mps', dtype=torch.float32, attn_implementation='sdpa',
)
wavs, sr = model.generate_custom_voice(
    text='Hello from the M4 Mac mini.',
    language='English', speaker='Ryan',
    instruct='Speak clearly with a calm, confident delivery.',
    max_new_tokens=512,
)
import soundfile as sf, numpy as np
sf.write('out.wav', (np.clip(wavs[0],-1,1)*32767).astype('int16'), sr)
"
