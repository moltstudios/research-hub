💻 Implementation Pattern (Next.js 16)

Three files: a service worker, a custom hook, and a server action for sending pushes via VAPID.

📁 public/sw.js — Service Worker

// public/sw.js — receives push events and shows notifications
self.addEventListener('push', (event) => {
  const data = event.data ? event.data.json() : {}
  const { title, body, url, icon } = data

  const options = {
    body,
    icon: icon || '/icon-192.png',
    badge: '/badge-72.png',
    data: { url },
    // Actions: rich push (Android/Chrome)
    actions: [
      { action: 'view', title: 'View' },
      { action: 'dismiss', title: 'Dismiss' }
    ],
    vibrate: [200, 100, 200],
    tag: 'cart-recovery',  // dedupe repeats
    renotify: true,
  }

  event.waitUntil(
    self.registration.showNotification(title, options)
  )
})

self.addEventListener('notificationclick', (event) => {
  event.notification.close()
  if (event.action === 'dismiss') return
  const url = event.notification.data?.url || '/'
  event.waitUntil(clients.openWindow(url))
})

📁 hooks/useWebPush.ts — Client subscription

// hooks/useWebPush.ts — soft prompt → native dialog → save subscription
import { useState } from 'react'

export function useWebPush() {
  const [status, setStatus] = useState<'idle' | 'subscribed' | 'denied'>('idle')

  const subscribe = async () => {
    if (!('serviceWorker' in navigator) || !('PushManager' in window)) {
      return false
    }

    const reg = await navigator.serviceWorker.register('/sw.js')
    const perm = await Notification.requestPermission()
    if (perm !== 'granted') { setStatus('denied'); return false }

    const sub = await reg.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: process.env.NEXT_PUBLIC_VAPID_PUBLIC
    })

    // Save to server
    await fetch('/api/push/subscribe', {
      method: 'POST',
      body: JSON.stringify(sub),
      headers: { 'Content-Type': 'application/json' }
    })

    setStatus('subscribed')
    return true
  }

  return { status, subscribe }
}

📁 app/api/push/send/route.ts — Server trigger (cart abandonment cron)

// app/api/push/send/route.ts — Vercel Cron triggers this on cart abandonment
import webpush from 'web-push'
import { createClient } from '@supabase/supabase-js'

webpush.setVapidDetails(
  'mailto:hello@opendrones.app',
  process.env.VAPID_PUBLIC!,
  process.env.VAPID_PRIVATE!
)

export async function POST(req: Request) {
  const { userId, title, body, url } = await req.json()
  const supabase = createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_KEY!)

  // Look up this user's push subscription(s)
  const { data: subs } = await supabase
    .from('push_subscriptions')
    .select('*')
    .eq('user_id', userId)

  // Fire-and-track
  await Promise.all(subs.map(s =>
    webpush.sendNotification(s.subscription, JSON.stringify({
      title, body, url, icon: '/icon-192.png'
    })).catch(async (err) => {
      if (err.statusCode === 410) {
        // Subscription expired — clean up
        await supabase.from('push_subscriptions').delete().eq('id', s.id)
      }
    })
  ))

  return Response.json({ sent: subs.length })
}

✅ Best Practices vs Pitfalls

✅ Do

  • Use a custom soft prompt before the native dialog (2-step flow = +600%)
  • Trigger on exit intent or after 30s of engagement (not on page load)
  • Show a clear value prop in the soft prompt ("Get restock alerts & flash sales")
  • Limit pushes to 1–2 per week to avoid the 46% opt-out cliff
  • Use the tag field to dedupe repeat notifications
  • Track click → conversion with UTM params per push

❌ Don't

  • Show the native dialog immediately on page load — kills opt-in rate
  • Ask for permission on the first page visit — no trust yet
  • Send more than 5 pushes per week — drives opt-outs
  • Use generic copy like "Subscribe to our newsletter" — vague kills CTR
  • Forget iOS PWA — push only works if the user installs to home screen
  • Skip subscription cleanup — dead subs (HTTP 410) pile up
