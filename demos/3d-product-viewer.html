📦 Scene Setup

// Basic Three.js scene initialization
import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(
  45,
  window.innerWidth / window.innerHeight,
  0.1, 1000
);
camera.position.set(0, 2, 5);

const renderer = new THREE.WebGLRenderer({
  antialias: true,
  alpha: true,
  powerPreference: 'high-performance'
});
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.outputColorSpace = THREE.SRGBColorSpace;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
document.body.appendChild(renderer.domElement);

🕹️ OrbitControls & Camera

// Smooth camera controls with damping
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.autoRotate = true;
controls.autoRotateSpeed = 1.5;
controls.minDistance = 2;
controls.maxDistance = 10;
controls.maxPolarAngle = Math.PI / 1.8;

// Animation loop with controls update
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
animate();

📂 GLB / GLTF Model Loading

// Load a product model from GLB file
const loader = new GLTFLoader();

const loadProduct = async (url) => {
  const { scene: model } = await loader.loadAsync(url);

  // Center and scale the model
  const box = new THREE.Box3().setFromObject(model);
  const center = box.getCenter(new THREE.Vector3());
  const size = box.getSize(new THREE.Vector3());
  const maxDim = Math.max(size.x, size.y, size.z);
  const scale = 2 / maxDim;

  model.position.sub(center);
  model.scale.multiplyScalar(scale);
  model.traverse((child) => {
    if (child.isMesh) {
      child.castShadow = true;
      child.receiveShadow = true;
    }
  });

  scene.add(model);
  return model;
};

💡 Lighting Setup for Products

// Studio-quality lighting for product shots
// Key light
const keyLight = new THREE.DirectionalLight('#ffffff', 1.2);
keyLight.position.set(5, 8, 5);
keyLight.castShadow = true;
scene.add(keyLight);

// Fill light (softer, opposite side)
const fillLight = new THREE.DirectionalLight('#b4c6ef', 0.5);
fillLight.position.set(-5, 3, -5);
scene.add(fillLight);

// Rim / back light for edge definition
const rimLight = new THREE.PointLight('#6366f1', 0.8, 20);
rimLight.position.set(0, 5, -5);
scene.add(rimLight);

// Ambient for base illumination
scene.add(new THREE.AmbientLight('#ffffff', 0.3));

// Optional: HDRI environment for realistic reflections
import { RGBELoader } from 'three/examples/jsm/loaders/RGBELoader';
const envMap = await new RGBELoader().loadAsync('/studio.hdr');
scene.environment = envMap;

⚡ Performance Tips

Instanced Meshes

Use InstancedMesh when rendering multiple copies of the same geometry — draw calls drop from N to 1.

Level of Detail (LOD)

Swap high-poly models for low-poly versions as objects move further from camera using THREE.LOD.

Lazy Load 3D Assets

Show a low-res placeholder or 2D image first. Load the full GLB only when the user interacts with the viewer.

Texture Compression

Use KTX2/Basis Universal textures for 6-8x smaller files with GPU-native decompression.

Dispose Resources

Always call .dispose() on geometries, materials, and textures when switching products or unmounting.

WebGL Renderer Hints

Set powerPreference: 'high-performance' and limit pixel ratio with Math.min(devicePixelRatio, 2).
