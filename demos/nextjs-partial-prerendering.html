💻 Implementation Code

next.config.ts TypeScript 
// Enable PPR in Next.js 15.3+/16
import type { Config } from 'next'

const config: Config = {
  experimental: {
    ppr: 'incremental'  // 'incremental' = opt-in per route
    // 'true' = enable for all routes (Next.js 16)
  }
}

export default config
app/products/[slug]/page.tsx TypeScript 
import { Suspense } from 'react'
import { getProduct } from '@/lib/db'
import { ProductHero } from '@/components/product-hero'
import { Inventory } from '@/components/inventory'
import { Reviews } from '@/components/reviews'

// Opt this route into PPR
export const experimental_ppr = true

// Static metadata — pre-rendered in shell
export async function generateMetadata({ params }) {
  const product = await getProduct(params.slug)
  return {
    title: product.name,
    description: product.description,
  }
}

export default async function ProductPage({ params }) {
  // This data is fetched at build time for the static shell
  const product = await getProduct(params.slug)

  return (
    <div>
      {/* STATIC: Pre-rendered in shell, cached at CDN */}
      <ProductHero product={product} />

      {/* DYNAMIC HOLE: Suspense boundary creates a "hole"
          in the static shell. Content streams when ready. */}
      <Suspense fallback={<InventorySkeleton />}>
        <Inventory productId={product.id} />
      </Suspense>

      <Suspense fallback={<ReviewsSkeleton />}>
        <Reviews productId={product.id} />
      </Suspense>
    </div>
  )
}
components/inventory.tsx TypeScript 
// This component fetches LIVE data — it becomes a dynamic hole
// The Suspense boundary around it creates the PPR "hole"

export async function Inventory({ productId }: { productId: string }) {
  // noStore() or dynamic data fetch = this is the "hole"
  // The rest of the page is static and pre-rendered
  const inventory = await getInventory(productId)

  return (
    <div className="inventory-badge">
      <span className="stock-count">{inventory.count} in stock</span>
      <span className="ship-time">Ships in {inventory.shipDays} days</span>
    </div>
  )
}

// Fallback skeleton shown while streaming
export function InventorySkeleton() {
  return (
    <div className="inventory-badge animate-pulse">
      <div className="h-4 w-24 bg-gray-200 rounded" />
      <div className="h-4 w-20 bg-gray-200 rounded" />
    </div>
  )
}
